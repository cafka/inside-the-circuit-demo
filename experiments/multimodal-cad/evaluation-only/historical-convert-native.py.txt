"""Official STEP -> actual OCCT triangulation -> intermediate GLB with CAD product tree."""
import sys
from pathlib import Path
import hashlib
import json
import time

HERE = Path(__file__).resolve().parent
ROOT = HERE.parent.parent
sys.path.insert(0, str(ROOT / '.local/cad-native'))
from OCP.STEPCAFControl import STEPCAFControl_Reader
from OCP.TDocStd import TDocStd_Document
from OCP.TCollection import TCollection_ExtendedString, TCollection_AsciiString
from OCP.XCAFDoc import XCAFDoc_DocumentTool
from OCP.TDF import TDF_LabelSequence
from OCP.IFSelect import IFSelect_RetDone
from OCP.BRepMesh import BRepMesh_IncrementalMesh
from OCP.BRep import BRep_Tool
from OCP.TopExp import TopExp_Explorer
from OCP.TopAbs import TopAbs_FACE
from OCP.TopoDS import TopoDS
from OCP.TopLoc import TopLoc_Location
from OCP.RWGltf import RWGltf_CafWriter
from OCP.RWMesh import RWMesh_CoordinateSystem_Zup, RWMesh_CoordinateSystem_Yup, RWMesh_NameFormat_Product
from OCP.TColStd import TColStd_IndexedDataMapOfStringString
from OCP.Message import Message_ProgressRange

CACHE = ROOT / '.local/cad-viewer'
source = CACHE / 'official.step'
expected = '78164070c1cc7ab854382950d2e85926c6e35432f1e5a146b733397a7c6afb44'
assert hashlib.sha256(source.read_bytes()).hexdigest() == expected
start = time.perf_counter()
doc = TDocStd_Document(TCollection_ExtendedString('BinXCAF'))
reader = STEPCAFControl_Reader()
reader.SetColorMode(True)
reader.SetNameMode(True)
print('Native OpenCascade 7.9.3: reading official STEP', flush=True)
assert reader.ReadFile(str(source)) == IFSelect_RetDone, 'STEP read failure'
read_end = time.perf_counter()
print('Native STEP read done; transferring named assemblies', flush=True)
assert reader.Transfer(doc), 'STEP document transfer failure'
transfer_end = time.perf_counter()
shape_tool = XCAFDoc_DocumentTool.ShapeTool_s(doc.Main())
roots = TDF_LabelSequence()
shape_tool.GetFreeShapes(roots)
faces = triangles = missing = 0
print('Tessellating native CAD surfaces', flush=True)
for i in range(1, roots.Length() + 1):
    shape = shape_tool.GetShape_s(roots.Value(i))
    mesher = BRepMesh_IncrementalMesh(shape, 0.05, False, 0.3, True)
    assert mesher.IsDone(), 'Tessellation failure'
    explorer = TopExp_Explorer(shape, TopAbs_FACE)
    while explorer.More():
        face = TopoDS.Face_s(explorer.Current())
        mesh = BRep_Tool.Triangulation_s(face, TopLoc_Location())
        faces += 1
        if mesh is None or mesh.NbTriangles() == 0:
            missing += 1
        else:
            triangles += mesh.NbTriangles()
        explorer.Next()
mesh_end = time.perf_counter()
assert missing == 0, f'{missing} CAD faces failed to triangulate'
print(f'Exporting {faces} faces / {triangles} triangles', flush=True)
writer = RWGltf_CafWriter(TCollection_AsciiString(str(CACHE / 'native-full.glb')), True)
writer.SetNodeNameFormat(RWMesh_NameFormat_Product)
writer.SetMeshNameFormat(RWMesh_NameFormat_Product)
writer.SetMergeFaces(True)
converter = writer.ChangeCoordinateSystemConverter()
converter.SetInputLengthUnit(0.001)
converter.SetOutputLengthUnit(1.0)
converter.SetInputCoordinateSystem(RWMesh_CoordinateSystem_Zup)
converter.SetOutputCoordinateSystem(RWMesh_CoordinateSystem_Yup)
assert writer.Perform(doc, TColStd_IndexedDataMapOfStringString(), Message_ProgressRange()), 'GLB export failure'
end = time.perf_counter()
report = dict(converter='OpenCascade 7.9.3 / cadquery-ocp 7.9.3.1.1 / RWGltf_CafWriter', sourceSHA256=expected,
              parameters=dict(linearUnit='millimeter', linearDeflectionType='absolute_value', linearDeflection=0.05, angularDeflection=0.3),
              importTimingMs=dict(read=(read_end-start)*1000, transfer=(transfer_end-read_end)*1000, triangulation=(mesh_end-transfer_end)*1000, nativeGLB=(end-mesh_end)*1000),
              cadFaces=faces, missingFaces=missing, nativeTriangles=triangles, assemblyRoots=roots.Length(),
              abandonedWasmAttempt='occt-import-js 0.0.23: interrupted after over 12 minutes without producing a mesh; approximately 2.3 GB private memory. No geometry from that incomplete attempt is delivered.')
(CACHE / 'native-source.json').write_text(json.dumps(report, indent=2), encoding='utf-8')
print(json.dumps(report, indent=2), flush=True)
